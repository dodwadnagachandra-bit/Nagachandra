    ```python
    async def _heartbeat_publisher(self) -> None:
        """Publish heartbeat to {prefix}/status at configured interval."""
        start_time: float = time.monotonic()
        heartbeat_interval: int = 60  # seconds, could be made configurable
        while not self._stop_event.is_set():
            await asyncio.sleep(heartbeat_interval)
            if not self._publisher.connected:
                continue
            payload: dict = {
                "device_id": self._prefix.split("/")[-1],  # e.g., "RES-001" from "ems/RES-001"
                "uptime_s": int(time.monotonic() - start_time),
                "version": __version__,
                "connected": True,
                "ts": int(time.time() * 1000),
            }
            self._publisher.publish_heartbeat(payload)
    ```
