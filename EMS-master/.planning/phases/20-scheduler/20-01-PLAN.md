    ```python
    from dataclasses import dataclass

    @dataclass(frozen=True, slots=True)
    class WindowResult:
        index: int | None      # Index of matched window, None if no match
        action: str            # "charge", "discharge", or "idle"
        power_kw: float        # Signed: negative=charge, positive=discharge, 0=idle

    @dataclass(frozen=True, slots=True)
    class CurveResult:
        index: int             # 0-95 curve index
        power_kw: float        # Signed value from power_curve array

    def parse_time(time_str: str) -> tuple[int, int]:
        """Parse 'HH:MM' string to (hour, minute)."""

    def evaluate_time_of_day(hour: int, minute: int, time_windows: list[dict]) -> WindowResult:
        """Find matching window for current time. First match wins. Half-open [start, end).
        Handles midnight-wrapping windows (start > end).
        Returns idle with power_kw=0 when outside all windows.
        For charge action: negate power_kw (negative=charge per convention).
        For discharge action: keep power_kw positive.
        For idle action: power_kw=0."""

    def evaluate_curve(hour: int, minute: int, power_curve: list[float]) -> CurveResult:
        """Calculate 96-point index = hour*4 + minute//15. Return CurveResult with signed value from array."""

    def evaluate_day_night(hour: int, minute: int, day_night: dict) -> str:
        """Return 'day' or 'night' based on current time and day_night config.
        day_night has day_start and night_start as 'HH:MM' strings.
        If current time >= day_start and < night_start -> 'day', else 'night'."""
    ```
