    ```python
    class AlarmEvaluator:
        def __init__(self, instances: dict[str, AlarmInstance]) -> None:
            self._instances = instances

        def evaluate_tick(self, values: dict[str, float | None], now_ms: float) -> list[dict]:
            """Evaluate all alarm rules for one tick. Returns list of event dicts."""
            events: list[dict] = []
            for alarm_id, inst in self._instances.items():
                if not inst.enabled:
                    continue
                value = values.get(inst.signal)
                tick_events = self._evaluate_one(inst, value, now_ms)
                events.extend(tick_events)
            return events

        def acknowledge(self, alarm_id: str) -> dict:
            """Acknowledge an alarm. Returns result dict with status, from_state, to_state."""
            ...

        def get_active_alarms(self) -> list[dict]:
            """Return non-NORMAL alarm instances as dicts."""
            ...

        def get_alarm_config(self) -> list[dict]:
            """Return all alarm rules as config dicts."""
            ...
    ```

    **_evaluate_one logic per state:**
    - NORMAL: check if exceeds threshold. If yes, start/continue delay timer. If delay expired, transition to ACTIVE_UNACKED. If not exceeding, reset delay timer.
    - ACTIVE_UNACKED: if signal clears (below clear threshold with hysteresis) -> CLEARED_UNACKED. Otherwise stay.
    - ACTIVE_ACKED: if signal clears -> RTN. Otherwise stay.
    - CLEARED_UNACKED: if signal re-exceeds threshold (no delay needed for re-activation from CLEARED_UNACKED) -> ACTIVE_UNACKED. Otherwise stay, wait for acknowledge.
    - RTN: auto-transition to NORMAL, reset all timestamps.

    **Event dict format:**
    ```python
    {
        "event_type": "alarm_activated",  # or alarm_acknowledged, alarm_cleared, alarm_rtn
        "alarm_id": inst.alarm_id,
        "signal": inst.signal,
        "severity": inst.severity,
        "state": inst.state,
        "value": inst.current_value,
        "threshold": inst.high_threshold or inst.low_threshold,
    }
    ```

    **Threshold check helper:**
    ```python
    def _check_exceeds(inst: AlarmInstance, value: float) -> bool:
        if inst.high_threshold is not None and value > inst.high_threshold:
            return True
        if inst.low_threshold is not None and value < inst.low_threshold:
            return True
        return False

    def _check_cleared(inst: AlarmInstance, value: float) -> bool:
        if inst.high_threshold is not None and inst.high_clear is not None:
            if value >= inst.high_clear:
                return False  # still above clear threshold
        if inst.low_threshold is not None and inst.low_clear is not None:
            if value <= inst.low_clear:
                return False  # still below clear threshold
        return True  # cleared
    ```

    **Critical edge case for _check_cleared:** For alarms with ONLY a high_threshold, clearance means value < high_clear. For alarms with ONLY a low_threshold, clearance means value > low_clear. The function must handle both cases correctly based on which threshold is set.
