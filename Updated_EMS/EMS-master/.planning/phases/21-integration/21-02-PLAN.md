```python
class TestCommandFlow:
    """E2E: HMI REST -> ZMQ -> control_manager -> RTDB -> PCS simulator."""

    @pytest.fixture(scope="class")
    def m3_system(self): ...  # Full system fixture as described above

    @pytest.mark.asyncio
    async def test_e2e_command_flow(self, m3_system):
        """Step-by-step command flow per CONTEXT.md decision."""
        hmi_port = m3_system["hmi_http_port"]
        base_url = f"http://127.0.0.1:{hmi_port}"

        async with httpx.AsyncClient(base_url=base_url, timeout=10.0) as client:
            # Step 1: Health check (already verified by fixture ready_check)

            # Step 2: Login with operator PIN "1234"
            resp = await client.post("/api/auth/login", json={"pin": "1234"})
            assert resp.status_code == 200, f"Login failed: {resp.text}"
            token = resp.json()["token"]
            headers = {"Authorization": f"Bearer {token}"}

            # Step 3: POST /api/control/mode {target_state: "standby"}
            resp = await client.post("/api/control/mode", json={"target_state": "standby"}, headers=headers)
            assert resp.status_code == 200, f"Mode change failed: {resp.text}"
            body = resp.json()
            assert body["status"] == "ok"

            # Step 4: Verify PCS state via RTDB (wait up to 15s)
            state = wait_for_criteria(
                {"standby": check_control_state(STATE_STANDBY)},
                timeout=15.0,
            )
            assert state["standby"], "control_state did not reach STANDBY within 15s"

            # Step 5: POST /api/control/setpoint {power_kw: 15.0}
            resp = await client.post("/api/control/setpoint", json={"power_kw": 15.0}, headers=headers)
            assert resp.status_code == 200, f"Setpoint failed: {resp.text}"
            body = resp.json()
            assert body["status"] == "ok"

            # Step 6: Verify RTDB setpoint (read active_setpoint_kw)
            # Use attach_rtdb to read system.active_setpoint_kw
            import time as _time
            deadline = _time.monotonic() + 5.0
            setpoint_verified = False
            while _time.monotonic() < deadline:
                from ems_common.rtdb import attach_rtdb, detach_rtdb
                shm, rtdb = attach_rtdb()
                setpoint = float(rtdb.system.active_setpoint_kw)
                del rtdb; detach_rtdb(shm)
                if abs(setpoint - 15.0) < 0.1:
                    setpoint_verified = True
                    break
                _time.sleep(0.3)
            assert setpoint_verified, f"active_setpoint_kw={setpoint}, expected 15.0"

            # Step 7: GET /api/alarm/active (verify no alarms)
            resp = await client.get("/api/alarm/active", headers=headers)
            assert resp.status_code == 200, f"Alarm query failed: {resp.text}"
```

```python
class TestWebSocketReconnection:
    """WebSocket server-side recovery: kill hmi_server, verify reconnectable."""

    @pytest.fixture(scope="class")
    def m3_system(self): ...  # Full system with hmi_server as subprocess

    @pytest.mark.asyncio
    async def test_websocket_reconnect_after_server_kill(self, m3_system):
        """Kill hmi_server, wait 5s, restart, reconnect WebSocket within 10s."""
        hmi_port = m3_system["hmi_http_port"]
        ws_uri = f"ws://127.0.0.1:{hmi_port}/ws/telemetry"
        hmi: ModuleProcess = m3_system["modules"]["hmi_server"]

        # Step 1: Connect WebSocket, receive 3+ messages
        messages_before: list[dict] = []
        async with websockets.connect(ws_uri) as ws:
            for _ in range(3):
                raw = await asyncio.wait_for(ws.recv(), timeout=5.0)
                messages_before.append(json.loads(raw))
        assert len(messages_before) >= 3, "Pre-kill: insufficient messages"

        # Step 2: Kill hmi_server with SIGKILL
        hmi.kill(sig=signal.SIGKILL)

        # Wait for process to die
        deadline = time.monotonic() + 3.0
        while hmi.is_alive and time.monotonic() < deadline:
            time.sleep(0.1)
        assert not hmi.is_alive, "hmi_server did not die after SIGKILL"

        # Step 3: Wait 5 seconds (simulates systemd RestartSec=5)
        time.sleep(5.0)

        # Step 4: Restart hmi_server
        hmi.restart()

        # Verify health endpoint
        health_ok = wait_for_criteria(
            {"health": lambda: _check_http_health(hmi_port)},
            timeout=10.0,
        )
        assert health_ok["health"], "hmi_server health endpoint not responding after restart"

        # Step 5: Reconnect WebSocket within 10s
        reconnected = False
        reconnect_deadline = time.monotonic() + 10.0
        messages_after: list[dict] = []

        while time.monotonic() < reconnect_deadline:
            try:
                async with websockets.connect(ws_uri) as ws:
                    for _ in range(3):
                        raw = await asyncio.wait_for(ws.recv(), timeout=5.0)
                        messages_after.append(json.loads(raw))
                    reconnected = True
                    break
            except (ConnectionRefusedError, OSError, websockets.exceptions.WebSocketException):
                await asyncio.sleep(1.0)

        # Step 6: Verify data flowing on new connection
        assert reconnected, "Failed to reconnect WebSocket within 10s after restart"
        assert len(messages_after) >= 3, f"Post-restart: only {len(messages_after)} messages"
```
