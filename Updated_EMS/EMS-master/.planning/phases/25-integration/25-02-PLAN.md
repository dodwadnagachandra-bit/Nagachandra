```python
class MosquittoController:
    def __init__(self, host: str, port: int, config_path: Path):
        self.host = host
        self.port = port
        self._config_path = config_path
        self._proc: subprocess.Popen | None = None

    def start(self) -> None:
        self._proc = subprocess.Popen(
            ["mosquitto", "-c", str(self._config_path)],
            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
        )
        # Wait for broker ready (paho connect test, deadline 5s)

    def stop(self) -> None:
        if self._proc:
            self._proc.terminate()
            try: self._proc.wait(timeout=5)
            except: self._proc.kill(); self._proc.wait()
            self._proc = None

    def restart(self) -> None:
        self.stop()
        time.sleep(0.5)  # Let port release
        self.start()
```

```python
class MockPartitionBackend:
    """Duck-type replacement for PartitionBackend -- uses temp dirs, no real partitions."""
    def __init__(self, tmpdir: Path):
        self._tmpdir = tmpdir
        self._active = "A"
        self._boot_flag_swapped = False
        self._image_written: Path | None = None
        self._rebooted = False

    def get_active_partition(self) -> str:
        return self._active

    def get_standby_partition(self) -> str:
        return "B" if self._active == "A" else "A"

    async def write_image_to_standby(self, image_path: Path) -> None:
        dest = self._tmpdir / "standby_image"
        import shutil
        shutil.copy(image_path, dest)
        self._image_written = dest

    async def swap_boot_flag(self) -> None:
        self._boot_flag_swapped = True
        self._active = self.get_standby_partition() if not self._boot_flag_swapped else ("B" if self._active == "A" else "A")

    async def reboot(self) -> None:
        self._rebooted = True  # NO-OP -- do NOT call systemctl reboot!

    async def rollback(self) -> None:
        self._active = "A"  # Revert to original
```
